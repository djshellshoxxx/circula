def git(*args):

def doc_git(*args):

def write_json(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def pr_span(numbers):
    ordered = sorted(numbers)
    if len(ordered) > 1 and ordered == list(range(ordered[0], ordered[-1] + 1)):
        return f'#{ordered[0]}–#{ordered[-1]}'
    return ', '.join(f'#{number}' for number in ordered)


def api_pr_head(pr):
    head = pr.get('head')
    return head.get('sha') if isinstance(head, dict) else head


def api_pr_branch(pr):
    head = pr.get('head')
    return pr.get('branch') or (head.get('ref') if isinstance(head, dict) else None)


def api_pr_base(pr):
    base = pr.get('base')
    return base.get('ref') if isinstance(base, dict) else base
