def retain(source, target):
    original = (ROOT / source).read_bytes()
    # Preserve log contents while making any whitespace-only normalization explicit.
    # Work on bytes: preserve CRLF, Unicode line separators and non-ASCII
    # whitespace. Only ASCII spaces/tabs at LF-delimited line ends and the
    # final LF count are normalized, with both hashes retained when changed.
    retained = b'\n'.join(line.rstrip(b' \t') for line in original.split(b'\n')).rstrip(b'\n') + b'\n' if original else b''
    transformation = None
    if target.endswith('.diff'):
        # A unified diff's context marker followed by a Go indentation tab is
        # meaningful patch data, not indentation in a new source file. Retain
        # the exact UTF-8 patch in a reversible JSON wrapper for inspection.
        records.pop(target, None)
        (OUT / target).unlink(missing_ok=True)
        target += '.json'
        payload = {'media_type': 'text/x-diff', 'original_utf8': original.decode('utf-8')}
        retained = (json.dumps(payload, indent=2, ensure_ascii=False) + '\n').encode('utf-8')
        assert json.loads(retained)['original_utf8'].encode('utf-8') == original
        transformation = {'kind': 'utf8_patch_json_wrapper', 'decode': 'Read original_utf8 and encode as UTF-8 to recover the exact original patch bytes.'}
    elif target in TYPED_DIGEST_JSON:
        # Public Git object identities are evidence, not credentials. Explicit
        # typing avoids ambiguous auth/password/API key-to-hash JSON pairs;
        # scanners and their scopes remain unchanged.
        source_value = json.loads(original)
        def typed(value):
            if isinstance(value, str) and re.fullmatch(r'[0-9a-f]{40}', value):
                return {'git_object_sha1': value}
            if isinstance(value, str) and re.fullmatch(r'[0-9a-f]{64}', value):
                return {'sha256_digest': value}
            if isinstance(value, dict):
                return {key: typed(item) for key, item in value.items()}
            if isinstance(value, list):
                return [typed(item) for item in value]
            return value
        def untyped(value):
            if isinstance(value, dict) and set(value) == {'git_object_sha1'}:
                return value['git_object_sha1']
            if isinstance(value, dict) and set(value) == {'sha256_digest'}:
                return value['sha256_digest']
            if isinstance(value, dict):
                return {key: untyped(item) for key, item in value.items()}
            if isinstance(value, list):
                return [untyped(item) for item in value]
            return value
        retained = (json.dumps(typed(source_value), indent=2, ensure_ascii=False) + '\n').encode('utf-8')
        assert untyped(json.loads(retained)) == source_value
        transformation = {'kind': 'typed_digest_values', 'decode': 'Recursively replace sole-key git_object_sha1 and sha256_digest objects with their string values to reconstruct the original JSON data. All 26 SHA-1 values in these three reviewed inputs were verified to resolve to Git objects. SHA-256 values retain their source-declared checksum meaning. Original byte hashes are retained separately; no scanner exception is introduced.'}
    (OUT / target).parent.mkdir(parents=True, exist_ok=True)
    (OUT / target).write_bytes(retained)
    entry = {'retained_path': target, 'source_paths': [source],
             'sha256': hashlib.sha256(retained).hexdigest(), 'bytes': len(retained)}
    if retained != original:
        entry['source_sha256'] = hashlib.sha256(original).hexdigest()
        if transformation:
            entry['transformation'] = transformation
        else:
            entry['normalization'] = 'Trailing ASCII spaces/tabs on LF-delimited lines and the final LF count only; all other bytes and line order unchanged.'
    records[target] = entry
