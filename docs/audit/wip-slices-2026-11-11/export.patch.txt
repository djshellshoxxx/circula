+// bigLedger appends n synthetic, pairwise balanced rows to the network-wide
+// entries table, so a large export costs no postings. Ids are random hex, so

+func (t bigTx) StreamTable(tbl store.ExportTable, netID, memberID string, fn func([]string, []any) error) error {
+	var cols []string
+	var last []any
+	err := t.Tx.StreamTable(tbl, netID, memberID, func(c []string, v []any) error {
+		cols, last = c, v
+		return fn(c, v)
+	})
+	if err != nil || tbl != store.TblEntries || memberID != "" || last == nil {
+		return err
+	}
+	rng := rand.New(rand.NewSource(1))
+	id := func() string { return fmt.Sprintf("%016x-%016x", rng.Uint64(), rng.Uint64()) }
+	for i := 0; i < t.n; i++ {
+		row := append([]any(nil), last...)
+		amt := int64(1 + i/2%997)
+		if i%2 == 1 {
+			amt = -amt
+		}
+		for j, c := range cols {
+			switch c {
+			case "id", "tx_id", "account_id":
+				row[j] = id()
+			case "amount":
+				row[j] = amt
+			}
+		}
+		if err := fn(cols, row); err != nil {
+			return err
+		}
+	}
+	return nil
+}

+// peakHeap runs fn and returns the largest growth of the live heap over its
+// starting point, sampled every millisecond with the collector made eager.
+func peakHeap(fn func()) uint64 {
+	defer debug.SetGCPercent(debug.SetGCPercent(5))
+	runtime.GC()
+	var ms runtime.MemStats
+	runtime.ReadMemStats(&ms)
+	base, peak := ms.HeapAlloc, ms.HeapAlloc
+	done := make(chan struct{})
+	var wg sync.WaitGroup
+	wg.Add(1)
+	go func() {
+		defer wg.Done()
+		var m runtime.MemStats
+		for {
+			runtime.ReadMemStats(&m)
+			peak = max(peak, m.HeapAlloc)
+			select {
+			case <-done:
+				return
+			case <-time.After(time.Millisecond):
+			}
+		}
+	}()
+	fn()
+	close(done)
+	wg.Wait()
+	return peak - base
+}

+// spoolFile is an export archive on disk (FR-095). It is written once while
+// the export streams and read with ReadAt for every download, so neither step
+// holds the archive in memory. Where the platform allows it (POSIX), the name
+// is unlinked as soon as the file exists: the open descriptor keeps the data
+// readable and nothing is left on disk if the process stops. Elsewhere the
+// file is removed when its export is dropped.
+type spoolFile struct {
+	f    *os.File
+	path string // name still on disk, removed on release; "" once unlinked
+	size int64
+}
+
+func (s *spoolFile) release() {
+	_ = s.f.Close()
+	if s.path != "" {
+		_ = os.Remove(s.path)
+	}
+}
+
+// artefactWriter receives the archive as the export produces it, hashing and
+// counting the bytes on the way. It spools to a new temporary file in dir
+// ("" is os.TempDir()); when none can be created it collects the archive in
+// memory, the DEC-047 interim behaviour.
+type artefactWriter struct {
+	spool *spoolFile
+	mem   bytes.Buffer
+	h     hash.Hash
+	n     int64
+}
+
+func newArtefactWriter(dir string) *artefactWriter {
+	a := &artefactWriter{h: sha256.New()}
+	f, err := os.CreateTemp(dir, "circula-export-*.zip") // mode 0600
+	if err != nil {
+		return a
+	}
+	a.spool = &spoolFile{f: f, path: f.Name()}
+	if os.Remove(f.Name()) == nil {
+		a.spool.path = ""
+	}
+	return a
+}
+
+func (a *artefactWriter) Write(p []byte) (int, error) {
+	var n int
+	var err error
+	if a.spool != nil {
+		n, err = a.spool.f.Write(p)
+	} else {
+		n, err = a.mem.Write(p)
+	}
+	a.h.Write(p[:n])
+	a.n += int64(n)
+	return n, err
+}
+
+// discard releases what was written; the writer is not used again.
+func (a *artefactWriter) discard() {
+	if a.spool != nil {
+		a.spool.release()
+	}
+	a.mem = bytes.Buffer{}
+}
+
+// finish hands the complete archive to e and returns its hex SHA-256.
+func (a *artefactWriter) finish(e *exportEntry) string {
+	if a.spool != nil {
+		a.spool.size = a.n
+		e.spool = a.spool
+	} else {
+		e.file = a.mem.Bytes()
+	}
+	return hex.EncodeToString(a.h.Sum(nil))
+}
+
+func (e *exportEntry) size() int64 {
+	if e.spool != nil {
+		return e.spool.size
+	}
+	return int64(len(e.file))
+}
+
+// release frees the artefact once its export has left the store.
+func (e *exportEntry) release() {
+	if e.spool != nil {
+		e.spool.release()
+	}
+}
+
+// content reads the artefact from its start; each call is independent, so
+// downloads may run side by side.
+func (e *exportEntry) content() *io.SectionReader {
+	if e.spool != nil {
+		return io.NewSectionReader(e.spool.f, 0, e.spool.size)
+	}
+	return io.NewSectionReader(bytes.NewReader(e.file), 0, int64(len(e.file)))
+}

+	for id, s := range perAcct {
+		if s != 0 {
+			ids = append(ids, id)
+		}
+	}
+	sort.Strings(ids)
+	h := sha256.New()

+// exportCorpus builds a network with seeded random trades, partial and full
+// reversals, adjustments and decided disputes (QA-030 property corpus).
+// Refused commands are part of the corpus; internal errors fail the test.
+func exportCorpus(t *testing.T, seed int64, steps int) (*world, []domain.Actor, []*domain.Member) {
+	t.Helper()
+	rng := rand.New(rand.NewSource(seed))
+	w := newWorldCfg(t, func(p *domain.Policy) { p.DefaultCreditLimit = 5000 }, func(c *service.Config) { c.ExportsPerHour = -1 })
+	rev, _ := w.reviewer()
+	actors, members := []domain.Actor{w.a, w.b}, []*domain.Member{w.aM, w.bM}
+	for i := 0; i < 3; i++ {
+		a, m := w.Member2(w.admin, fmt.Sprintf("c%d@example.org", i))
+		actors, members = append(actors, a), append(members, m)
+	}
+	check := func(err error, what string) {
+		if err != nil && domain.CodeOf(err) == domain.CodeInternal {
+			t.Fatalf("seed %d %s: %v", seed, what, err)
+		}
+	}
+	completed := func() *domain.Transaction {
+		l := tu.Must(w.Svc.AdminListTransactions(w.Ctx, w.admin, w.net.ID, "", []domain.TxStatus{domain.TxCompleted}, service.Page{Limit: 200}))
+		if len(l) == 0 {
+			return nil
+		}
+		return l[rng.Intn(len(l))]
+	}
+	for step := 0; step < steps; step++ {
+		switch rng.Intn(7) {
+		case 0, 1, 2: // trade
+			i, j := rng.Intn(len(members)), rng.Intn(len(members))
+			tx, err := w.Svc.ProposeTransaction(w.Ctx, actors[i], w.net.ID, service.ProposeInput{CounterpartyID: members[j].ID,
+				Direction: service.ProposerPays, Amount: 1 + rng.Int63n(2000), IdempotencyKey: w.Key()})
+			check(err, "propose")
+			if err == nil {
+				_, err = w.Svc.AcceptTransaction(w.Ctx, actors[j], w.net.ID, service.TxRef{TxID: tx.ID, IdempotencyKey: w.Key()})
+				check(err, "accept")
+			}
+		case 3: // full or partial reversal
+			if tx := completed(); tx != nil && tx.Kind != domain.KindReversal {
+				amt := int64(0)
+				if rng.Intn(2) == 0 {
+					amt = 1 + rng.Int63n(tx.Amount)
+				}
+				_, err := w.Svc.ReverseTransaction(w.Ctx, w.admin, w.net.ID, service.ReverseInput{TxID: tx.ID, Amount: amt, Reason: "r", IdempotencyKey: w.Key()})
+				check(err, "reverse")
+			}
+		case 4: // adjustment
+			dir := service.AdjustCredit
+			if rng.Intn(2) == 0 {
+				dir = service.AdjustDebit
+			}
+			_, err := w.Svc.CreateAdjustment(w.Ctx, w.admin, w.net.ID, service.AdjustInput{MemberID: members[rng.Intn(len(members))].ID,
+				Amount: 1 + rng.Int63n(3000), Direction: dir, Reason: "r", IdempotencyKey: w.Key()})
+			check(err, "adjust")
+		case 5, 6: // dispute, decided at once
+			tx := completed()
+			if tx == nil || tx.Kind != domain.KindTrade {
+				break
+			}
+			payer := actors[0]
+			for i, m := range members {
+				if m.ID == tx.PayerMemberID {
+					payer = actors[i]
+				}
+			}
+			d, err := w.Svc.OpenDispute(w.Ctx, payer, w.net.ID, service.OpenDisputeInput{TxID: tx.ID, Category: "other", IdempotencyKey: w.Key()})
+			check(err, "dispute")
+			if err != nil {
+				break
+			}
+			outcome := []domain.DisputeOutcome{domain.OutcomeDismissed, domain.OutcomeUpheld, domain.OutcomePartiallyUpheld}[rng.Intn(3)]
+			amt := int64(0)
+			if outcome == domain.OutcomePartiallyUpheld {
+				amt = 1 + rng.Int63n(tx.Amount)
+			}
+			_, err = w.Svc.ResolveDispute(w.Ctx, rev, w.net.ID, service.ResolveInput{DisputeID: d.ID, Outcome: outcome, Amount: amt, Rationale: "r", IdempotencyKey: w.Key()})
+			check(err, "resolve")
+		}
+	}
+	return w, actors, members
+}

+// pauseStore runs hook (once) inside an export's read transaction, after the
+// accounts and before the entries are read, so a test can commit postings
+// while the export is in flight.
+type pauseStore struct {
+	store.Store
+	hook atomic.Pointer[func()]
+}
+
+func (p *pauseStore) View(ctx context.Context, fn func(store.Tx) error) error {
+	return p.Store.View(ctx, func(tx store.Tx) error { return fn(pauseTx{Tx: tx, p: p}) })
+}
+
+type pauseTx struct {
+	store.Tx
+	p *pauseStore
+}
+
+func (t pauseTx) StreamTable(tbl store.ExportTable, netID, memberID string, fn func([]string, []any) error) error {
+	if tbl == store.TblEntries {
+		if h := t.p.hook.Swap(nil); h != nil {
+			(*h)()
+		}
+	}
+	return t.Tx.StreamTable(tbl, netID, memberID, fn)
+}

+			}
+		}
+	}

+	wg.Wait()

+		t.Fatalf("poster: %v", err)

+	}
