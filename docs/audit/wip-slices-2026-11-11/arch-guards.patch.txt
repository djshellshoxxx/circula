+package store_test
+
+import (
+	"os"
+	"path/filepath"
+	"regexp"
+	"sort"
+	"strconv"
+	"strings"
+	"testing"
+)
+
+// DAT-045: schema changes follow expand-then-contract. A migration may add
+// structures; removing or renaming one (the contract step) is only allowed in
+// a later release, after the reads moved off it. Every migration that drops or
+// renames anything must therefore be listed here, with the earlier migration
+// and release after which nothing reads the old structure. Any other DROP or
+// RENAME fails the build. Migrations 0001..0015 contain none.
+var contractSteps = map[string]string{
+	// "00NN_name.sql": "reads moved to <new structure> in 00MM (release vX.Y); <what is dropped or renamed>",
+}
+
+// contractRE matches the statements that remove or rename schema objects:
+// DROP TABLE/INDEX/TRIGGER/VIEW, ALTER TABLE ... DROP COLUMN and
+// ALTER TABLE ... RENAME [COLUMN] ... TO.
+var contractRE = regexp.MustCompile(`(?i)\b(drop|rename)\b`)
+
+// sqlCode blanks comments and quoted strings or identifiers, keeping newlines,
+// so a keyword in a comment or a RAISE message is not taken for a statement.
+func sqlCode(src string) string {
+	b := []byte(src)
+	blank := func(i int) {
+		if b[i] != '\n' {
+			b[i] = ' '
+		}
+	}
+	for i := 0; i < len(b); i++ {
+		switch {
+		case b[i] == '-' && i+1 < len(b) && b[i+1] == '-':
+			for ; i < len(b) && b[i] != '\n'; i++ {
+				blank(i)
+			}
+		case b[i] == '/' && i+1 < len(b) && b[i+1] == '*':
+			for ; i < len(b); i++ {
+				end := b[i] == '*' && i+1 < len(b) && b[i+1] == '/'
+				blank(i)
+				if end {
+					i++
+					blank(i)
+					break
+				}
+			}
+		case b[i] == '\'' || b[i] == '"' || b[i] == '`':
+			q := b[i]
+			blank(i)
+			for i++; i < len(b); i++ {
+				if b[i] == q {
+					if i+1 < len(b) && b[i+1] == q { // doubled quote escapes itself
+						blank(i)
+						i++
+						blank(i)
+						continue
+					}
+					blank(i)
+					break
+				}
+				blank(i)
+			}
+		}
+	}
+	return string(b)
+}
+
+// contractStatements returns "line N: <source line>" for every DROP or RENAME
+// outside comments and strings.
+func contractStatements(src string) []string {
+	code := sqlCode(src)
+	lines := strings.Split(src, "\n")
+	var out []string
+	for i, l := range strings.Split(code, "\n") {
+		if contractRE.MatchString(l) {
+			out = append(out, "line "+strconv.Itoa(i+1)+": "+strings.TrimSpace(lines[i]))
+		}
+	}
+	return out
+}
+
+func TestMigrationsExpandThenContract(t *testing.T) {
+	files, err := filepath.Glob(filepath.Join("migrations", "*.sql"))
+	if err != nil || len(files) == 0 {
+		t.Fatalf("no migrations found: %v", err)
+	}
+	sort.Strings(files)
+	seen := map[string]bool{}
+	for _, f := range files {
+		name := filepath.Base(f)
+		seen[name] = true
+		src, err := os.ReadFile(f)
+		if err != nil {
+			t.Fatal(err)
+		}
+		found := contractStatements(string(src))
+		_, allowed := contractSteps[name]
+		switch {
+		case len(found) > 0 && !allowed:
+			t.Errorf("%s drops or renames schema objects without a contract-step entry (DAT-045: expand, backfill, switch reads, "+
+				"then contract in a later release; list it in contractSteps with the migration that moved the reads):\n  %s",
+				name, strings.Join(found, "\n  "))
+		case len(found) == 0 && allowed:
+			t.Errorf("contractSteps lists %s but it neither drops nor renames anything; remove the entry", name)
+		}
+	}
+	for name := range contractSteps {
+		if !seen[name] {
+			t.Errorf("contractSteps lists %s, which is not a migration", name)
+		}
+	}
+}
+
+// The scanner itself: every contract form is caught, and keywords inside
+// comments, strings or longer identifiers are not.
+func TestContractScannerForms(t *testing.T) {
+	caught := []string{
+		"DROP TABLE accounts;",
+		"drop index idx_x;",
+		"DROP TRIGGER IF EXISTS entries_no_update;",
+		"DROP VIEW v;",
+		"ALTER TABLE persons DROP COLUMN login;",
+		"ALTER TABLE persons RENAME TO people;",
+		"ALTER TABLE persons RENAME COLUMN login TO handle;",
+		"CREATE TABLE t (a INT); /* c */ Drop Table t;",
+	}
+	for _, s := range caught {
+		if len(contractStatements(s)) == 0 {
+			t.Errorf("not caught: %q", s)
+		}
+	}
+	ignored := []string{
+		"-- we never DROP or RENAME anything here",
+		"/* DROP TABLE x;\n RENAME */ CREATE TABLE y (a INT);",
+		"SELECT RAISE(ABORT, 'cannot drop: append-only');",
+		`CREATE TABLE "drop" (a INT);`,
+		"CREATE TABLE t (dropped_at INTEGER, drop_reason TEXT, renamed INTEGER);",
+		"SELECT 'it''s DROP';",
+	}
+	for _, s := range ignored {
+		if got := contractStatements(s); len(got) != 0 {
+			t.Errorf("false positive for %q: %v", s, got)
+		}
+	}
+	if got := contractStatements("CREATE TABLE a (x INT);\n-- note\nDROP TABLE a;"); len(got) != 1 || !strings.HasPrefix(got[0], "line 3:") {
+		t.Errorf("line reporting = %v", got)
+	}
+}

+// concurrentUnitsSerialise: N concurrent read-check-write units against one
+// account whose bound admits only k of them commit exactly k (ARC-006), and
+// concurrent counter increments never repeat a value.
+func concurrentUnitsSerialise(t *testing.T, s store.Store) {
+	f := seed(t, s, "a")
+	const n, step, floor = 50, 10, -100 // floor/step = 10 may pass
+	var wg sync.WaitGroup
+	var mu sync.Mutex
+	committed, seqs := 0, map[int64]bool{}
+	errs := make(chan error, n)
+	for i := 0; i < n; i++ {
+		wg.Add(1)
+		go func() {
+			defer wg.Done()
+			var seq int64
+			err := s.Update(ctx, func(tx store.Tx) error {
+				acc, err := tx.GetAccount(f.net, f.accA)
+				if err != nil {
+					return err
+				}
+				if acc.Balance-step < floor {
+					return errAbort
+				}
+				if seq, err = tx.NextLedgerSeq(f.net); err != nil {
+					return err
+				}
+				if err := tx.AddBalance(f.net, f.accA, -step); err != nil {
+					return err
+				}
+				return tx.AddBalance(f.net, f.sys, step)
+			})
+			switch {
+			case err == nil:
+				mu.Lock()
+				committed++
+				if seqs[seq] {
+					errs <- fmt.Errorf("ledger sequence %d handed out twice", seq)
+				}
+				seqs[seq] = true
+				mu.Unlock()
+			case errors.Is(err, errAbort):
+			default:
+				errs <- err
+			}
+		}()
+	}
+	wg.Wait()
+	close(errs)
+	for err := range errs {
+		t.Error(err)
+	}
+	if want := -floor / step; committed != want {
+		t.Errorf("committed %d units, want exactly %d", committed, want)
+	}
+	if got := balance(t, s, f.net, f.accA); got != floor {
+		t.Errorf("balance = %d, want %d", got, floor)
+	}
+	if got := balance(t, s, f.net, f.sys); got != -floor {
+		t.Errorf("system balance = %d, want %d (postings are equal and opposite)", got, -floor)
+	}
+}

+			}

+						if s, ok := constStringExpr(vs.Values[i], consts); ok {
+							consts[n.Name] = s
+						}
+					}
+				}
+			}
+		}
+	}
+	return files, consts

+func routeCallee(fun ast.Expr) (recv, name string) {
+	switch f := fun.(type) {
+	case *ast.Ident:
+		return "", f.Name
+	case *ast.SelectorExpr:
+		switch x := f.X.(type) {
+		case *ast.Ident:
+			return x.Name, f.Sel.Name
+		case *ast.SelectorExpr:
+			return x.Sel.Name, f.Sel.Name
+		}
+		return "?", f.Sel.Name
+	}
+	return "", ""
+}

+// routePlumbing are the registration calls that forward a pattern held in a
+// variable; every other registration must use a constant pattern.
+var routePlumbing = map[string]bool{
+	"api/route.go:add":     true, // s.mux.HandleFunc(rt.pattern, s.wrap(rt))
+	"api/routes.go:routes": true, // R's body: s.add(rt)

+// scanRoutes lists the routes registered by the non-test files of dir.
+func scanRoutes(t *testing.T, surface, dir string) []scannedRoute {
+	t.Helper()
+	fset := token.NewFileSet()
+	files, consts := parseGoDir(t, fset, dir)
+	var out []scannedRoute
+	for _, f := range files {
+		file := filepath.Base(fset.Position(f.Pos()).Filename)
+		for _, d := range f.Decls {
+			site := surface + "/" + file + ":" // package-level declarations (function literals in vars)
+			if fd, ok := d.(*ast.FuncDecl); ok {
+				site += fd.Name.Name
+			}
+			ast.Inspect(d, func(n ast.Node) bool {
+				call, ok := n.(*ast.CallExpr)
+				if !ok || len(call.Args) == 0 {
+					return true
+				}
+				recv, name := routeCallee(call.Fun)
+				isReg := false
+				switch name {
+				case "HandleFunc", "Handle":
+					isReg = true
+					if recv != "mux" {
+						t.Errorf("%s: route registered on %q, not on a mux the inventory reads", fset.Position(call.Pos()), recv)
+						return true
+					}
+				case "R":
+					isReg = recv == "" && surface == "api"
+				case "add":
+					isReg = recv == "s" && surface == "api"
+				}
+				if !isReg {
+					return true
+				}
+				arg := call.Args[0]
+				if cl, ok := arg.(*ast.CompositeLit); ok && name == "add" {
+					arg = nil
+					for _, el := range cl.Elts {
+						if kv, ok := el.(*ast.KeyValueExpr); ok {
+							if k, ok := kv.Key.(*ast.Ident); ok && k.Name == "pattern" {
+								arg = kv.Value
+							}
+						}
+					}
+					if arg == nil {
+						t.Errorf("%s: route literal without a pattern", fset.Position(call.Pos()))
+						return true
+					}
+				}
+				pat, ok := constStringExpr(arg, consts)
+				if !ok {
+					if !routePlumbing[site] {
+						t.Errorf("%s: route pattern is not a constant; the inventory cannot read it (register with a constant pattern)",
+							fset.Position(call.Pos()))
+					}
+					return true
+				}
+				m, p := splitRoutePattern(pat)
+				out = append(out, scannedRoute{routeKey{surface, m, p}, derivedGate(surface, name, p, call), fset.Position(call.Pos()).String()})
+				return true
+			})
+		}
+	}
+	return out
+}

+// derivedGate reads the gate the registration itself enforces.
+func derivedGate(surface, name, path string, call *ast.CallExpr) string {
+	switch surface {
+	case "api":
+		if name != "R" {
+			return "public" // the not-found fallback
+		}
+		gate := "member"
+		for _, a := range call.Args[2:] {
+			if id, ok := a.(*ast.Ident); ok {
+				switch id.Name {
+				case "noNet":
+					gate = "person"
+				case "public":
+					return "public"
+				}
+			}
+		}
+		return gate
+	case "web":
+		if len(call.Args) < 2 {
+			return "public"
+		}
+		if c, ok := call.Args[1].(*ast.CallExpr); ok {
+			if _, n := routeCallee(c.Fun); n != "" {
+				switch n {
+				case "auth":
+					return "person"
+				case "netH":
+					return "member"
+				case "adminH":
+					return "admin"
+				}
+			}
+		}
+		return "public"
+	default: // ops
+		if path == "/" || path == "/api/" {
+			return "mount"
+		}
+		return "health"
+	}
+}

+// gateMatches reports whether the table's gate is compatible with what the
+// registration enforces. Roles finer than membership are decided by the
+// service (api) or the handler (web reviewer pages).
+func gateMatches(surface, derived, table, path string) bool {
+	if surface == "api" && strings.HasPrefix(path, v1+"/admin/") && table != "admin" {
+		return false
+	}
+	switch derived {
+	case "member":
+		return table == "member" || table == "reviewer" || (table == "admin" && surface == "api")
+	default:
+		return table == derived
+	}
+}

+func readAuthzTable(t *testing.T) []authzRow {
+	t.Helper()
+	f, err := os.Open(authzTablePath)
+	if err != nil {
+		t.Fatal(err)
+	}
+	defer func() { _ = f.Close() }()
+	var rows []authzRow
+	sc := bufio.NewScanner(f)
+	for n := 1; sc.Scan(); n++ {
+		line := sc.Text()
+		if i := strings.Index(line, "#"); i >= 0 {
+			line = line[:i]
+		}
+		fs := strings.Fields(line)
+		if len(fs) == 0 {
+			continue
+		}
+		if len(fs) != 4 {
+			t.Fatalf("%s:%d: want 4 fields (surface method path gate), got %q", authzTablePath, n, sc.Text())
+		}
+		rows = append(rows, authzRow{routeKey{fs[0], fs[1], fs[2]}, fs[3], n})
+	}
+	if err := sc.Err(); err != nil {
+		t.Fatal(err)
+	}
+	return rows
+}

+// concretePath turns a route pattern into a request path.
+func concretePath(path string, vals map[string]string) string {
+	path = strings.ReplaceAll(path, "{$}", "")
+	var b strings.Builder
+	for {
+		i := strings.Index(path, "{")
+		if i < 0 {
+			b.WriteString(path)
+			return b.String()
+		}
+		j := strings.Index(path[i:], "}")
+		b.WriteString(path[:i])
+		name := strings.TrimSuffix(path[i+1:i+j], "...")
+		if v, ok := vals[name]; ok {
+			b.WriteString(v)
+		} else {
+			b.WriteString("x")
+		}
+		path = path[i+j+1:]
+	}
+}

+// TestRouteAuthorizationMatrix: anonymous callers, people of no network and

+// refusal for a network the caller does not belong to looks like not-found

+func TestRouteAuthorizationMatrix(t *testing.T) {

+	srv := apiH.(*server)
+
+	token := func(a domain.Actor) string {
+		raw, _, err := env.Svc.CreateAPIToken(env.Ctx, a, service.TokenInput{Name: "t", Scopes: []domain.Scope{domain.ScopeRead, domain.ScopeWrite}})

+		}
+		return raw
+	}

+		if err != nil {
+			t.Fatal(err)
+		}

+	sessions := map[string]string{"alice": login("alice"), "outsider": login("outsider")}
+	apiVals := map[string]string{"role": "reviewer"}
+	webVals := map[string]string{"slug": net.Slug}

+	api := map[string][]routeProbe{

+		"member":   {{"anonymous", func(s int) bool { return s == 401 }, "401"}, {"outsider", func(s int) bool { return s == 404 }, "404 not-found"}},
+		"reviewer": {{"anonymous", func(s int) bool { return s == 401 }, "401"}, {"outsider", func(s int) bool { return s == 404 }, "404 not-found"}, {"alice", is4xx, "a 4xx refusal"}},
+		"admin":    {{"anonymous", func(s int) bool { return s == 401 }, "401"}, {"outsider", func(s int) bool { return s == 404 }, "404 not-found"}, {"alice", is4xx, "a 4xx refusal"}},
+	}
+	toLogin := func(s int) bool { return s == http.StatusSeeOther }
+	webP := map[string][]routeProbe{
+		"person":   {{"anonymous", toLogin, "303 to /login"}},
+		"member":   {{"anonymous", toLogin, "303 to /login"}, {"outsider", func(s int) bool { return s == 404 }, "404"}},
+		"reviewer": {{"anonymous", toLogin, "303 to /login"}, {"outsider", func(s int) bool { return s == 404 }, "404"}, {"alice", func(s int) bool { return s == 403 }, "403"}},
+		"admin":    {{"anonymous", toLogin, "303 to /login"}, {"outsider", func(s int) bool { return s == 404 }, "404"}, {"alice", func(s int) bool { return s == 403 }, "403"}},

+				if pr.who == "anonymous" && rec.Code == http.StatusSeeOther && !strings.HasPrefix(rec.Header().Get("Location"), "/login") {

+				}
+			}
+		}
+	}

+func probeRoute(t *testing.T, r authzRow, pr routeProbe, h http.Handler, hr *http.Request) *httptest.ResponseRecorder {
+	t.Helper()
+	rec := httptest.NewRecorder()
+	h.ServeHTTP(rec, hr)
+	if !pr.deny(rec.Code) || rec.Code >= 500 {
+		what := "unexpected response"
+		if rec.Code < 400 && rec.Code != http.StatusSeeOther {
+			what = "unexpected allow"
+		}
+		t.Errorf("%s (gate %s) as %s: status %d, want %s (%s; %s:%d)", r.routeKey, r.gate, pr.who, rec.Code, pr.want,
+			what, authzTablePath, r.line)
+	}
+	if pr.who == "outsider" && r.surface == "api" && rec.Code == 404 && !strings.Contains(rec.Body.String(), `"code":"not-found"`) {
+		t.Errorf("%s as outsider: 404 body %s is not the plain not-found problem", r.routeKey, rec.Body.String())
+	}
+	return rec
+}
