+// maxReversalLinks bounds one ReversalsOf answer; it is the store's page maximum.
+const maxReversalLinks = 500
+
+// ReversalsOf returns the reversals posted against transaction txID, oldest first, so
+// an original line can link to what reversed it (UX-017). It only reads. Whoever may
+// see the original (GetTransaction) may see the list; anyone else, and any id from
+// another network, gets the same not-found as a missing id (SEC-009).
+func (s *Service) ReversalsOf(ctx context.Context, actor domain.Actor, netID, txID string) ([]*domain.Transaction, error) {
+	var out []*domain.Transaction
+	err := s.view(ctx, func(tx store.Tx) error {
+		p, err := s.open(tx, actor, netID, domain.ScopeRead)
+		if err != nil {
+			return err
+		}
+		t, err := tx.GetTransaction(netID, txID)
+		if err != nil {
+			return domain.NotFound("transaction")
+		}
+		if ok, err := canSeeTx(tx, p, t); err != nil {
+			return err
+		} else if !ok {
+			return domain.NotFound("transaction")
+		}
+		out, err = tx.ListTransactions(netID, store.TxFilter{OriginalTxID: t.ID, Kind: domain.KindReversal, Limit: maxReversalLinks})
+		return err
+	})
+	if err != nil {
+		return nil, err
+	}
+	sort.SliceStable(out, func(i, j int) bool {
+		if out[i].LedgerSeq != out[j].LedgerSeq {
+			return out[i].LedgerSeq < out[j].LedgerSeq
+		}
+		return out[i].ID < out[j].ID
+	})
+	return out, nil
+}

+// memberRef is the stable member reference shown beside a display name in confirmation
+// steps (SEC-016): display names are not unique, member ids are and never change. It is the
+// last eight letters or digits of the id, upper-cased and split in two ("0A1B-2C3D"), or ""
+// when there is no member (the network's own account).
+func memberRef(id string) string {
+	var b []byte
+	for i := 0; i < len(id); i++ {
+		if ch := id[i]; ch >= '0' && ch <= '9' || ch >= 'a' && ch <= 'z' || ch >= 'A' && ch <= 'Z' {
+			b = append(b, ch)
+		}
+	}
+	if len(b) > 8 {
+		b = b[len(b)-8:]
+	}
+	ref := strings.ToUpper(string(b))
+	if len(ref) == 8 {
+		ref = ref[:4] + "-" + ref[4:]
+	}
+	return ref
+}

+// rowOf returns the History list item whose first link is href.
+func rowOf(body, href string) string {

+	}

+// domainCodes parses the domain package and returns every declared Code value, so a code
+// added later without a web mapping fails the table test below.
+func domainCodes(t *testing.T) []domain.Code {
+	t.Helper()
+	files, err := filepath.Glob(filepath.Join("..", "domain", "*.go"))
+	if err != nil || len(files) == 0 {
+		t.Fatalf("domain sources: %v", err)
+	}
+	fset := token.NewFileSet()
+	seen := map[domain.Code]bool{}
+	for _, f := range files {
+		if strings.HasSuffix(f, "_test.go") {
+			continue
+		}
+		file, err := parser.ParseFile(fset, f, nil, 0)
+		if err != nil {
+			t.Fatal(err)
+		}
+		for _, d := range file.Decls {
+			g, ok := d.(*ast.GenDecl)
+			if !ok || g.Tok != token.CONST {
+				continue
+			}
+			for _, s := range g.Specs {
+				vs := s.(*ast.ValueSpec)
+				typed := false
+				if id, ok := vs.Type.(*ast.Ident); ok && id.Name == "Code" {
+					typed = true
+				}
+				for _, v := range vs.Values {
+					lit, ok := v.(*ast.BasicLit)
+					if call, isCall := v.(*ast.CallExpr); isCall {
+						if fn, ok2 := call.Fun.(*ast.Ident); ok2 && fn.Name == "Code" && len(call.Args) == 1 {
+							lit, ok = call.Args[0].(*ast.BasicLit)
+							typed = ok
+						}
+					}
+					if !typed || !ok || lit.Kind != token.STRING {
+						continue
+					}
+					c, err := strconv.Unquote(lit.Value)
+					if err != nil {
+						t.Fatal(err)
+					}
+					seen[domain.Code(c)] = true
+				}
+			}
+		}
+	}
+	out := make([]domain.Code, 0, len(seen))
+	for c := range seen {
+		out = append(out, c)
+	}
+	sort.Slice(out, func(i, j int) bool { return out[i] < out[j] })
+	return out
+}

+// helpFooter pins the shared Help link to one place: the first link of the second
+// paragraph of the page footer, after the main landmark.
+var helpFooter = regexp.MustCompile(`</main>\s*<footer class="foot">\s*<p>[^<]*</p>\s*<p><a href="/help">Help: credits, balances and statuses</a>`)

+func assertHelpLocation(t *testing.T, where, body string) {
+	t.Helper()
+	if !helpFooter.MatchString(body) {
+		t.Errorf("%s: the Help link is not in its footer location", where)
+	}
+	foot := body[strings.LastIndex(body, `<footer class="foot">`)+1:]
+	if n := strings.Count(foot, `href="/help"`); n != 1 {
+		t.Errorf("%s: %d Help links in the footer, want 1", where, n)
+	}
+	if i := strings.Index(body, "<main"); i >= 0 && strings.Contains(body[:i], `href="/help`) {
+		t.Errorf("%s: a Help link moved into the header", where)
+	}
