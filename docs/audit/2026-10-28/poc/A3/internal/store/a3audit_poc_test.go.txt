package store_test

// A3 audit proof-of-concept (2026-10-28). Throwaway evidence.

import (
	"context"
	"path/filepath"
	"testing"

	"github.com/djshellshoxxx/circula/internal/store"
)

// A3-F12 (info): migrateTo validates recorded names with all[v-1] and only
// checks v > len(all); a schema_migrations row with version <= 0 (a crafted
// or damaged database: an -allow-unsigned restore, or file access) makes
// every open panic instead of returning an error.
func TestA3Audit_MigrationVersionZeroPanics(t *testing.T) {
	ctx := context.Background()
	path := filepath.Join(t.TempDir(), "c.db")
	s, err := store.Open(ctx, path)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := s.DB().ExecContext(ctx, `INSERT INTO schema_migrations(version,name,applied_at) VALUES (0,'x',0)`); err != nil {
		t.Fatal(err)
	}
	_ = s.Close()
	defer func() {
		if p := recover(); p != nil {
			t.Errorf("store.Open panicked on a schema_migrations row with version 0: %v", p)
		}
	}()
	s2, err := store.Open(ctx, path)
	if err == nil {
		_ = s2.Close()
	}
}
