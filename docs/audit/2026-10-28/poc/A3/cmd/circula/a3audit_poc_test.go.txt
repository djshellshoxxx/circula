// A3-F01: with backup-encryption-key configured (the secure, recommended
// setting, and the default-enforced one via backup-require-encryption=auto),
// `circula migrate` must still be able to take and verify its pre-migration
// backup and migrate the database.
func TestA3Audit_MigrateWithEncryptionKey(t *testing.T) {
	dir := t.TempDir()
	path := seedV1Database(t, dir)
	keyFile := writeKeyFile(t, dir, "bk.key", testKey(t))

	var out, errOut bytes.Buffer
	code := run([]string{"migrate", "-database", path, "-backup-encryption-key-file", keyFile}, &out, &errOut)
	if code != exitOK {
		t.Fatalf("migrate with backup-encryption-key failed: code %d\nstdout: %s\nstderr: %s", code, out.String(), errOut.String())
	}
}

// A3-F01 (unit level): preMigrationSafeguard on a current database with an
// encryption key.
func TestA3Audit_PreMigrationSafeguardEncrypted(t *testing.T) {
	ctx := context.Background()
	db := seedDatabase(t, t.TempDir())
	cfg := encConfig(db, testKey(t))
	cfg.BackupRequireEncryption = true
	state, err := inspectSchema(ctx, db)
	if err != nil {
		t.Fatal(err)
	}
	if _, p, err := preMigrationSafeguard(ctx, cfg, state); err != nil {
		t.Fatalf("pre-migration safeguard refused an encrypted backup it just wrote: %v (file %s)", err, p)
	}
}

// A3-F09: a restore brings back every session and API token that existed at
// backup time, including ones revoked afterwards (logout, password change,
// "revoke token" after a leak). Nothing in restore revokes them (SA-07's
// "revoke all on restore" suggestion is not implemented).
func TestA3Audit_RestoreResurrectsRevokedSession(t *testing.T) {
	ctx := context.Background()
	dir := t.TempDir()
	db := seedDatabase(t, dir)
	open := func() (*store.SQLite, *service.Service) {
		st, err := store.Open(ctx, db)
		if err != nil {
			t.Fatal(err)
		}
		svc, err := service.New(st, service.Config{})
		if err != nil {
			t.Fatal(err)
		}
		return st, svc
	}
	st, svc := open()
	res, err := svc.LoginStep(ctx, "founder01", "correct horse battery")
	if err != nil {
		t.Fatal(err)
	}
	tok := res.Token
	_ = st.Close()

	keyFile := writeKeyFile(t, dir, "sign.key", testKey(t))
	bk := filepath.Join(dir, "bk")
	if err := os.Mkdir(bk, 0o700); err != nil {
		t.Fatal(err)
	}
	var out, errOut bytes.Buffer
	if code := run([]string{"backup", "-database", db, "-backup-key-file", keyFile, bk}, &out, &errOut); code != exitOK {
		t.Fatalf("backup: %d %s", code, errOut.String())
	}
	ents, _ := os.ReadDir(bk)
	var file string
	for _, e := range ents {
		if strings.HasSuffix(e.Name(), ".db") {
			file = filepath.Join(bk, e.Name())
		}
	}

	st, svc = open()
	if err := svc.Logout(ctx, tok); err != nil { // the person signs out (or the session is revoked after a leak)
		t.Fatal(err)
	}
	if _, err := svc.Authenticate(ctx, tok); err == nil {
		t.Fatal("control: revoked session still authenticates")
	}
	_ = st.Close()

	out.Reset()
	errOut.Reset()
	if code := run([]string{"restore", "-database", db, "-backup-key-file", keyFile, "-overwrite", file}, &out, &errOut); code != exitOK {
		t.Fatalf("restore: %d %s %s", code, out.String(), errOut.String())
	}
	st, svc = open()
	defer func() { _ = st.Close() }()
	if a, err := svc.Authenticate(ctx, tok); err == nil {
		t.Errorf("session revoked after the backup authenticates again after restore (person %s)", a.PersonID)
	}
}

// A3-F10: the backup schedule and retention trust the timestamp in the file
// name. One backup written while the host clock ran ahead (NTP fault) stops
// every scheduled backup until that future time plus the interval, and it is
// always kept by retention. Nothing logs the anomaly (OPS-033 clock error).
func TestA3Audit_FutureDatedBackupStopsSchedule(t *testing.T) {
	f := newJobFixture(t)
	if err := os.MkdirAll(f.cfg.BackupDir, 0o700); err != nil {
		t.Fatal(err)
	}
	ahead := time.Now().UTC().AddDate(10, 0, 0) // written while the clock read 2036
	if err := os.WriteFile(filepath.Join(f.cfg.BackupDir, backupFileName(ahead)), []byte("x"), 0o600); err != nil {
		t.Fatal(err)
	}
	if d := f.runner.untilNextBackup(); d > f.cfg.BackupInterval {
		t.Errorf("after the clock was corrected the next scheduled backup waits %v (about %d days)", d.Round(time.Hour), int(d.Hours()/24))
	}
	// Retention: 3 real daily backups with keep-daily 2 -> the future file is kept, a real one is pruned.
	for i := 1; i <= 3; i++ {
		p := filepath.Join(f.cfg.BackupDir, backupFileName(time.Now().UTC().Add(-time.Duration(i)*24*time.Hour)))
		if err := os.WriteFile(p, []byte("x"), 0o600); err != nil {
			t.Fatal(err)
		}
	}
	removed, err := pruneBackups(f.cfg.BackupDir, 2, 0)
	if err != nil {
		t.Fatal(err)
	}
	if len(removed) != 2 {
		t.Logf("removed %v", removed)
	}
	left, _ := listBackups(f.cfg.BackupDir)
	if len(left) > 0 && left[0].at.After(time.Now().Add(24*time.Hour)) {
		t.Logf("retention keeps the future-dated file as the newest backup; real backups kept: %d", len(left)-1)
	}
}

// A3-F17: writeManifest (and writeJSONFile, writeAnchorFile) write to the
// fixed name <path>.tmp with os.WriteFile, which follows a symlink and
// truncates its target. `circula backup <dir>` into a directory another local
// user can write lets that user point the predictable
// circula-<UTC second>.db.manifest.json.tmp at any file the operator account
// can write; the manifest JSON replaces it.
func TestA3Audit_ManifestTmpFollowsSymlink(t *testing.T) {
	dir := t.TempDir()
	victim := filepath.Join(dir, "operator-owned.conf")
	if err := os.WriteFile(victim, []byte("important=1\n"), 0o600); err != nil {
		t.Fatal(err)
	}
	dest := filepath.Join(dir, "circula-20261028T120000Z.db")
	if err := os.Symlink(victim, dest+".manifest.json.tmp"); err != nil {
		t.Skip(err)
	}
	if err := writeManifest(dest+".manifest.json", &Manifest{Format: manifestFormat}); err != nil {
		t.Logf("writeManifest: %v", err)
	}
	b, _ := os.ReadFile(victim)
	if !bytes.Contains(b, []byte("important=1")) {
		t.Errorf("a file outside the backup was overwritten through the .tmp symlink; it now holds: %.60q", b)
	}
}
