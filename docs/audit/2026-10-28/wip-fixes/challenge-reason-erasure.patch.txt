+// A3-F05: erasure empties the reason of every challenge the member filed,
+// decided or still waiting, like the member's other free text
+// (docs/dev/erasure.md, "Adding personal data to the model"). The challenge
+// rows, the decider's rationale and other members' challenges stay as they are.
+func TestErasureClearsMemberChallengeReasons(t *testing.T) {
+	w, _ := erasureWorld(t)
+	rev, _ := w.reviewer()
+	const (
+		decidedText   = "Ana was removed unfairly, call 555-0100"
+		waitingText   = "please reconsider, write to ana.private@example.net"
+		otherText     = "my limit was cut without notice"
+		rationaleText = "no grounds were recorded"
+	)
+	// Another member's challenge, which the erasure must leave alone.
+	lim := int64(100)
+	tu.Must(0, w.Svc.SetCreditLimit(w.Ctx, w.admin, w.net.ID, service.SetLimitInput{MemberID: w.bM.ID, Limit: &lim, Reason: "risk"}))
+	other := tu.Must(w.Svc.OpenChallenge(w.Ctx, w.b, w.net.ID, service.ChallengeInput{Kind: service.ChallengeCreditLimit, Reason: otherText}))
+	// Ana is removed and challenges the removal twice: one challenge is decided, one still waits.
+	w.depart(w.admin, w.aM)
+	decided := tu.Must(w.Svc.OpenChallenge(w.Ctx, w.a, w.net.ID, service.ChallengeInput{Kind: service.ChallengeRemoval, Reason: decidedText}))
+	tu.Must(w.Svc.DecideChallenge(w.Ctx, rev, w.net.ID, service.ChallengeDecision{ID: decided.ID, Outcome: service.ChallengeRejected, Rationale: rationaleText}))
+	waiting := tu.Must(w.Svc.OpenChallenge(w.Ctx, w.a, w.net.ID, service.ChallengeInput{Kind: service.ChallengeRemoval, Reason: waitingText}))
+	for _, needle := range []string{"555-0100", "ana.private@example.net"} {
+		if len(dbMatches(t, w.Store.DB(), needle)) == 0 {
+			t.Fatalf("test premise: %q should be stored before erasure", needle)
+		}
+	}
+
+	rep := tu.Must(w.Svc.EraseMyData(w.Ctx, w.a, w.net.ID, ""))
+	if !rep.PersonErased {
+		t.Fatalf("person not erased: %+v", rep)
+	}
+	// The withdrawn dispute's statement and the two challenge reasons.
+	if rep.DisputeStatements != 3 {
+		t.Errorf("free-text fields cleared = %d, want 3", rep.DisputeStatements)
+	}
+	for _, needle := range []string{"555-0100", "ana.private@example.net", "unfairly", "reconsider", anaLogin, "Ana"} {
+		if hits := dbMatches(t, w.Store.DB(), needle); len(hits) > 0 {
+			t.Errorf("%q still found in %v", needle, hits)
+		}
+	}
+
+	// The challenges stay on record; only the member's own text is gone.
+	got := map[string]store.Challenge{}
+	for _, c := range tu.Must(w.Svc.ListChallenges(w.Ctx, rev, w.net.ID, "")) {
+		got[c.ID] = c
+	}
+	if c := got[decided.ID]; c.MemberID != w.aM.ID || c.Kind != service.ChallengeRemoval || c.Reason != "" ||
+		c.Status != "decided" || c.Outcome != service.ChallengeRejected || c.Rationale != rationaleText {
+		t.Errorf("decided challenge after erasure: %+v", c)
+	}
+	if c := got[waiting.ID]; c.MemberID != w.aM.ID || c.Kind != service.ChallengeRemoval || c.Reason != "" || c.Status != "requested" {
+		t.Errorf("waiting challenge after erasure: %+v", c)
+	}
+	if c := got[other.ID]; c.MemberID != w.bM.ID || c.Kind != service.ChallengeCreditLimit || c.Reason != otherText || c.Status != "requested" {
+		t.Errorf("another member's challenge changed: %+v", c)
+	}
+	if rep := tu.Must(w.Svc.VerifyLedger(w.Ctx, w.admin, w.net.ID)); !rep.OK {
+		t.Fatal(rep.Findings)
+	}
+}
