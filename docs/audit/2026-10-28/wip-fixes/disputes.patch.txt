+// GOV-020 (A1-04): with no active administrator the network is read-only and
+// no value moves. A dispute decision that would post a reversal is refused
+// with the error the re-review path gives in that state, and changes nothing.
+// A dismissal posts nothing and is not gated, as a confirmed re-review is not.
+// Once an administrator is active again the same decision posts.
+func TestDisputeDecisionPostsNothingWhenAdminRequired(t *testing.T) {
+	w := newWorld(t, nil)
+	rev1, _ := w.reviewerAs("rev1@example.org")
+	rev2, _ := w.reviewerAs("rev2@example.org")
+	tx := w.Trade(w.net.ID, w.a, w.b, w.aM, w.bM, 1000)
+	d := tu.Must(w.open(w.a, tx.ID))
+	// A decided dispute awaiting its re-review gives the re-review path's refusal.
+	rr := tu.Must(w.open(w.a, w.Trade(w.net.ID, w.a, w.b, w.aM, w.bM, 300).ID))
+	tu.Must(w.resolve(rev1, rr.ID, domain.OutcomeDismissed, 0, "no evidence"))
+	if err := w.rereview(w.a, rr.ID, "new evidence"); err != nil {
+		t.Fatal(err)
+	}
+	other := tu.Must(w.open(w.b, w.Trade(w.net.ID, w.a, w.b, w.aM, w.bM, 200).ID))
+
+	w.setFounderStatus(t, domain.MemberSuspended)
+	seq := w.balance(w.a).LedgerSeq
+	gate := w.decideRR(rev2, rr.ID, string(domain.OutcomeUpheld), 0)
+	wantCode(t, gate, domain.ErrInvalidState)
+	for _, c := range []struct {
+		outcome domain.DisputeOutcome
+		amount  int64
+	}{{domain.OutcomeUpheld, 0}, {domain.OutcomePartiallyUpheld, 400}} {
+		_, err := w.resolve(rev1, d.ID, c.outcome, c.amount, "not delivered")
+		wantCode(t, err, domain.ErrInvalidState)
+		if err.Error() != gate.Error() {
+			t.Errorf("%s refused with %q, want the re-review path's %q", c.outcome, err, gate)
+		}
+	}
+	v := tu.Must(w.Svc.GetDispute(w.Ctx, rev1, w.net.ID, d.ID))
+	if v.Dispute.Status != domain.DisputeOpen || v.Tx.Status != domain.TxDisputed || v.Tx.ReversedTotal != 0 ||
+		w.balance(w.a).LedgerSeq != seq {
+		t.Fatalf("refused decision changed state: dispute %s, trade %s reversed %d, ledger_seq %d -> %d",
+			v.Dispute.Status, v.Tx.Status, v.Tx.ReversedTotal, seq, w.balance(w.a).LedgerSeq)
+	}
+	if got := tu.Must(w.resolve(rev1, other.ID, domain.OutcomeDismissed, 0, "delivered")); got.Status != domain.DisputeResolved ||
+		w.balance(w.a).LedgerSeq != seq {
+		t.Fatalf("dismissal = %+v, ledger_seq %d -> %d", got, seq, w.balance(w.a).LedgerSeq)
+	}
+
+	w.setFounderStatus(t, domain.MemberActive)
+	got := tu.Must(w.resolve(rev1, d.ID, domain.OutcomeUpheld, 0, "not delivered"))
+	if got.Status != domain.DisputeResolved || w.txStatus(tx.ID).Status != domain.TxReversed || w.balance(w.a).LedgerSeq != seq+1 {
+		t.Fatalf("with an administrator again: dispute %s, trade %s, ledger_seq %d -> %d",
+			got.Status, w.txStatus(tx.ID).Status, seq, w.balance(w.a).LedgerSeq)
+	}
+}

+// GOV-020, DOM-021 (A1-04): while no administrator is active, an upheld
+// decision whose reversal LED-5 would also block is refused by the network
+// gate, not left posting_pending, and the retry of a posting_pending dispute
+// is refused although LED-5 would now allow it. Both change nothing. With an
+// administrator active the decision is held by LED-5 and the retry posts.
+func TestDisputeRetryPostsNothingWhenAdminRequired(t *testing.T) {
+	w := newWorld(t, func(p *domain.Policy) { p.DefaultCreditLimit = 2500 })
+	rev, _ := w.reviewer()

+	refused := func(err, gate error, want domain.DisputeStatus) {
+		t.Helper()
+		wantCode(t, err, domain.ErrInvalidState)
+		if err.Error() != gate.Error() {
+			t.Errorf("refused with %q, want the network gate's %q", err, gate)
+		}
+		v := tu.Must(w.Svc.GetDispute(w.Ctx, rev, w.net.ID, d.ID))
+		if v.Dispute.Status != want || v.Tx.Status != domain.TxDisputed || v.Tx.ReversedTotal != 0 || w.balance(w.a).LedgerSeq != seq {
+			t.Fatalf("refused call changed state: dispute %s, trade %s reversed %d, ledger_seq %d -> %d",
+				v.Dispute.Status, v.Tx.Status, v.Tx.ReversedTotal, seq, w.balance(w.a).LedgerSeq)
+		}
+	}

+// wantReversalAudited checks that the newest reversal of the trade orig has
+// exactly one audit event naming it: transaction.reversed by the decider, with
+// the reversal's own reason, both parties as subjects, links to the original
+// and the dispute, and the trade's state after the decision. A party finds the
+// event by the same id (FR-082).
+func wantReversalAudited(t *testing.T, w *world, orig, disputeID string, decider *domain.Member, amount int64) {
+	t.Helper()
+	var rev *domain.Transaction
+	for _, x := range tu.Must(w.Svc.AdminListTransactions(w.Ctx, w.admin, w.net.ID, domain.KindReversal, nil, service.Page{})) {
+		if x.OriginalTxID == orig && (rev == nil || x.LedgerSeq > rev.LedgerSeq) {
+			rev = x
+		}
+	}
+	if rev == nil || rev.Amount != amount {
+		t.Fatalf("newest reversal of %s = %+v, want amount %d", orig, rev, amount)
+	}
+	evs := tu.Must(w.Svc.ListAudit(w.Ctx, w.admin, w.net.ID, service.AuditQuery{TargetID: rev.ID}))
+	if len(evs) != 1 {
+		t.Fatalf("%d audit events name reversal %s, want 1", len(evs), rev.ID)
+	}
+	e, trade := evs[0], w.txStatus(orig)
+	parties := []string{w.aM.ID, w.bM.ID}
+	sort.Strings(parties)
+	if e.Action != domain.ActTxReversed || e.TargetType != "transaction" || e.ActorType != domain.ActorReviewer ||
+		e.ActorID != decider.ID || e.Reason != rev.Reason || !reflect.DeepEqual(e.SubjectIDs, parties) {
+		t.Errorf("reversal audit event = %+v", e)
+	}
+	wantSummary(t, "before", e.SummaryBefore, map[string]any{"original": orig, "dispute": disputeID})
+	wantSummary(t, "after", e.SummaryAfter, map[string]any{"amount": amount, "original_status": trade.Status, "reversed_total": trade.ReversedTotal})
+	if mine := tu.Must(w.Svc.ListAudit(w.Ctx, w.a, w.net.ID, service.AuditQuery{TargetID: rev.ID})); len(mine) != 1 || mine[0].Action != domain.ActTxReversed {
+		t.Errorf("the opener does not find the reversal by its id: %+v", mine)
+	}
+}
